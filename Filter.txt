youtube.com##html, body:style(overflow: auto !important; position: static !important;)

! 4. Neutralizes the JS trigger that renders this component and pauses the player
youtube.com##+js(set, ytInitialPlayerResponse.auxiliaryUi.messageRenderers.enforcementMessageViewModel, undefined)
